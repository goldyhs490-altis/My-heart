        // Firefly Class
        class Firefly {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * height;
                this.size = Math.random() * 2.5 + 1;
                this.speedX = (Math.random() - 0.5) * 0.6;
                this.speedY = (Math.random() - 0.5) * 0.6;
                this.alpha = Math.random();
                this.fadeSpeed = Math.random() * 0.015 + 0.005;
                this.color = Math.random() > 0.3 ? '#ffdd73' : '#ff9ebb'; // Warm yellow & soft pink
            }

            update() {
                this.x += this.speedX;
                this.y += this.speedY;

                // Bounce off boundaries gently
                if (this.x < 0 || this.x > width) this.speedX *= -1;
                if (this.y < 0 || this.y > height) this.speedY *= -1;

                // Pulsating brightness
                this.alpha += this.fadeSpeed;
                if (this.alpha >= 1 || this.alpha <= 0.1) {
                    this.fadeSpeed = -this.fadeSpeed;
                }
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = Math.max(0, this.alpha);
                ctx.shadowBlur = this.size * 5;
                ctx.shadowColor = this.color;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

        // Swirling Heart Class
        class FloatingHeart {
            constructor() {
                this.reset(true);
            }

            reset(initial = false) {
                this.x = Math.random() * width;
                this.y = initial ? Math.random() * height : height + 20;
                this.size = Math.random() * 14 + 10;
                this.speedY = Math.random() * 0.7 + 0.3;
                this.swayAngle = Math.random() * Math.PI * 2;
                this.swaySpeed = Math.random() * 0.02 + 0.01;
                this.opacity = Math.random() * 0.5 + 0.2;
                this.rotation = Math.random() * Math.PI * 2;
                this.rotSpeed = (Math.random() - 0.5) * 0.02;
                this.color = `rgba(${255}, ${Math.floor(Math.random() * 100 + 100)}, ${Math.floor(Math.random() * 150 + 120)}, ${this.opacity})`;
            }

            update() {
                this.y -= this.speedY;
                this.swayAngle += this.swaySpeed;
                this.x += Math.sin(this.swayAngle) * 0.5;
                this.rotation += this.rotSpeed;

                if (this.y < -30) {
                    this.reset(false);
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);
                ctx.fillStyle = this.color;
                ctx.shadowBlur = 8;
                ctx.shadowColor = 'rgba(255, 105, 180, 0.4)';

                // Draw Path SVG-like Heart
                ctx.beginPath();
                const topCurveHeight = this.size * 0.3;
                ctx.moveTo(0, topCurveHeight);
                // Top left curve
                ctx.bezierCurveTo(
                    0, 0,
                    -this.size / 2, 0,
                    -this.size / 2, topCurveHeight
                );
                // Bottom left curve
                ctx.bezierCurveTo(
                    -this.size / 2, (this.size + topCurveHeight) / 2,
                    0, this.size,
                    0, this.size
                );
                // Bottom right curve
                ctx.bezierCurveTo(
                    0, this.size,
                    this.size / 2, (this.size + topCurveHeight) / 2,
                    this.size / 2, topCurveHeight
                );
                // Top right curve
                ctx.bezierCurveTo(
                    this.size / 2, 0,
                    0, 0,
                    0, topCurveHeight
                );
                ctx.closePath();
                ctx.fill();
                ctx.restore();
            }
        }

                196.00, 246.94, 293.66, 392.00, 493.88  // G3, B3, D4, G4, B4
            ];

            let noteIdx = 0;

            function playNextNote() {
                if (!isAudioPlaying || !audioContext) return;

                const osc = audioContext.createOscillator();
                const noteGain = audioContext.createGain();

                // Soft sine wave for warm lullaby tone
                osc.type = 'sine';
                osc.frequency.setValueAtTime(notes[noteIdx % notes.length], audioContext.currentTime);

                // Gentle Envelope
                noteGain.gain.setValueAtTime(0.001, audioContext.currentTime);
                noteGain.gain.exponentialRampToValueAtTime(0.08, audioContext.currentTime + 0.3);
                noteGain.gain.exponentialRampToValueAtTime(0.0001, audioContext.currentTime + 2.5);

                osc.connect(noteGain);
                noteGain.connect(audioGain);

                osc.start();
                osc.stop(audioContext.currentTime + 2.6);

                noteIdx++;
                setTimeout(playNextNote, 800 + Math.random() * 400);
            }

            isAudioPlaying = true;
            playNextNote();
        }

        function toggleAudio() {
            if (!audioContext) {
                initAudio();
                return;
            }

            if (audioContext.state === 'suspended') {
                audioContext.resume();
                isAudioPlaying = true;
                audioIcon.className = 'fa-solid fa-music text-pink-300';
            } else if (isAudioPlaying) {
                audioGain.gain.setValueAtTime(0, audioContext.currentTime);
                isAudioPlaying = false;
                audioIcon.className = 'fa-solid fa-volume-xmark text-gray-400';
            } else {
                audioGain.gain.setValueAtTime(0.15, audioContext.currentTime);
                isAudioPlaying = true;
                playAmbientMelody();
                audioIcon.className = 'fa-solid fa-music text-pink-300';
            }
        }

        /* ===================================================
           5. SEQUENTIAL TEXT DISPLAY LOGIC (5s DELAY)
           =================================================== */
        function showLine(index) {
            if (index >= lines.length) {
                // Sequence completed
                finishSequence();
                return;
            }

            // Update Progress Bar
            const progressPercent = ((index + 1) / lines.length) * 100;
            progressBar.style.width = `${progressPercent}%`;

            // Fade Out current line
            textDisplay.classList.remove('active');
            textDisplay.classList.add('exit');

            setTimeout(() => {
                // Update text content
                textDisplay.textContent = lines[index];
                textDisplay.classList.remove('exit');

                // Trigger reflow for smooth transition restart
                void textDisplay.offsetWidth;

                // Fade In new line
                textDisplay.classList.add('active');
            }, 800); // 800ms fade transition window

            // Schedule next line
            if (!isPaused) {
                sequenceTimer = setTimeout(() => {
                    currentIndex++;
                    showLine(currentIndex);
                }, DISPLAY_DELAY);
            }
        }

        function startSequence() {
            currentIndex = 0;
            isPaused = false;
            finalCard.classList.remove('show');
            progressContainer.classList.add('visible');
            controlsBar.classList.add('visible');
            pauseIcon.className = 'fa-solid fa-pause text-pink-300';
            showLine(currentIndex);
        }

        function finishSequence() {
            clearTimeout(sequenceTimer);
            textDisplay.classList.remove('active');
           
            setTimeout(() => {
                textDisplay.textContent = '';
                finalCard.classList.add('show');
            }, 1000);
        }

        function togglePause() {
            if (isPaused) {
                isPaused = false;
                pauseIcon.className = 'fa-solid fa-pause text-pink-300';
                showLine(currentIndex);
            } else {
                isPaused = true;
                clearTimeout(sequenceTimer);
                pauseIcon.className = 'fa-solid fa-play text-pink-300';
            }
        }

        function restartSequence() {
            clearTimeout(sequenceTimer);
            startSequence();
        }

        /* ===================================================
           6. EVENT LISTENERS & INITIALIZATION
           =================================================== */
        function openCurtain() {
            if (curtainOpened) return;
            curtainOpened = true;

            // Open curtain animation
            curtainWrapper.classList.add('open');

            // Initialize ambient sound
            initAudio();

            // Hide curtain overlay after transition completes
            setTimeout(() => {
                curtainWrapper.style.visibility = 'hidden';
            }, 2800);

            // Start Text Sequence with a soft slight delay after curtain begins sliding
            setTimeout(() => {
                startSequence();
            }, 1200);
        }

        curtainCue.addEventListener('click', openCurtain);
        curtainWrapper.addEventListener('click', openCurtain);

        audioToggle.addEventListener('click', (e) => {
            e.stopPropagation();
            toggleAudio();
        });

        pauseToggle.addEventListener('click', (e) => {
            e.stopPropagation();
            togglePause();
        });

        restartBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            restartSequence();
        });

        replayBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            restartSequence();
        });

        // Start Canvas Particles on Page Load
        window.onload = () => {
            animateCanvas();
        };
    </script>
